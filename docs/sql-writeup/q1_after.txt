Pattern: Keyset Pagination (WHERE vin > cursor) + Composite Index + Single JOIN (Zero N+1)
Database: PostgreSQL 16 (Index: ix_vehicle_fleet_vin on vehicle(fleet_id, vin))

Query:
SELECT 
    v.vin,
    v.fleet_id,
    v.model_year,
    vm.name AS model_name,
    vm.battery_kwh,
    vm.rated_range_km,
    vm.max_dc_kw,
    se.soh_pct AS latest_soh_pct
FROM vehicle v
JOIN vehicle_model vm ON v.model_id = vm.model_id
LEFT JOIN LATERAL (
    SELECT soh_pct 
    FROM soh_estimate 
    WHERE vin = v.vin 
    ORDER BY est_at DESC 
    LIMIT 1
) se ON true
WHERE v.fleet_id = 1
  AND v.vin > '1VXEV01A100005000'
  AND v.erased_at IS NULL
ORDER BY v.vin ASC
LIMIT 50;
